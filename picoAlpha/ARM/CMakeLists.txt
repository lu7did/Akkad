cmake_minimum_required(VERSION 3.16)
include(pico_sdk_import.cmake)
if(NOT DEFINED PICO_BOARD)
 set(PICO_BOARD pico2_w CACHE STRING "Target board" FORCE)
endif()
if(NOT PICO_BOARD STREQUAL "pico2_w" AND NOT PICO_BOARD STREQUAL "pico_w")
 message(FATAL_ERROR "Supported boards: pico2_w (default) or pico_w (experimental)")
endif()
project(picoAlpha_firmware VERSION 0.2.54 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PICO_CXX_ENABLE_EXCEPTIONS 1)
pico_sdk_init()
if(PICO_BOARD STREQUAL "pico_w")
 message(WARNING "Pico W experimental: verificar margen de SRAM, heap y MCTS antes de uso prolongado.")
 add_compile_definitions(PICOALPHA_TARGET_PICO_W=1)
endif()
set(PICOALPHA_NET "1x8" CACHE STRING "Network selected at compile time: 1x8 or 2x16")
set_property(CACHE PICOALPHA_NET PROPERTY STRINGS 1x8 2x16)
if(PICOALPHA_NET STREQUAL "1x8")
  set(PICOALPHA_NET_SOURCE src/net_1x8.cpp)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/picoGyal.hpp" OR
     NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/picoGyal_fc_int8.hpp")
    message(FATAL_ERROR "Missing 1x8 INT8 coefficients: run tools/picoDistiller/picoExport.py first")
  endif()
elseif(PICOALPHA_NET STREQUAL "2x16")
  set(PICOALPHA_NET_SOURCE src/net.cpp)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/generated_tinygyal.hpp")
    message(FATAL_ERROR "Missing original TinyGyal 2x16 coefficients")
  endif()
else()
  message(FATAL_ERROR "Invalid PICOALPHA_NET: ${PICOALPHA_NET}")
endif()
message(STATUS "picoAlpha network: ${PICOALPHA_NET}")
# PA_SEGMENTED_TREE_BEGIN — v0.2.20: fail-closed, segmented by default.
# NEW rollback option deliberately avoids any stale PA_SEGMENTED_TREE=OFF cache.
option(PA_LEGACY_MCTS_TREE "Explicit rollback to monolithic tree (unsafe for long matches)" OFF)
if(PA_LEGACY_MCTS_TREE)
 set(PA_SEGMENTED_TREE OFF CACHE BOOL "Derived, do not edit" FORCE)
else()
 set(PA_SEGMENTED_TREE ON CACHE BOOL "Derived, do not edit" FORCE)
endif()
message(STATUS "picoAlpha v${PROJECT_VERSION} MCTS_STORAGE=${PA_SEGMENTED_TREE} (PA_LEGACY_MCTS_TREE=${PA_LEGACY_MCTS_TREE})")
# PA_SEGMENTED_TREE_END
# PA_NN_OPT_BEGIN: switchable network paths, default conservative FP32.
option(PA_TELEM_MCTS "MCTS guards and integrity telemetry" OFF)
option(PA_TELEM_MEM "Detailed fixed arena and growth telemetry" OFF)
option(PA_TELEM_NN "Neural compute telemetry" OFF)
option(PA_NN_PROFILE "Profile TinyGyal layers" ON)
option(PA_NN_WORKSPACE "Reusable TinyGyal buffers" ON)
option(PA_NN_FC_FAST "4-lane FP32 fully-connected kernel" ON)
option(PA_NN_FC_INT8 "Quantized INT8 FC1" ON)
set(PA_NN_FC_INT8 ON CACHE BOOL "FC1 INT8 default" FORCE)
option(PA_NN_DSP "Use Cortex-M DSP dot products when supported" ON)
option(PA_NN_CACHE "8-slot history-exact neural inference cache" ON)
option(PA_MCTS_REUSE "Reuse validated MCTS descendants across moves" ON)
option(PA_DUAL_CORE_NN "Split neural inference across both RP2350 cores" OFF)
set(PA_DUAL_CORE_NN OFF CACHE BOOL "Disabled for stability validation in v0.2.28" FORCE)
set(PA_STATIC_TREE_NODES 8576 CACHE STRING "MCTS static SRAM nodes, 20 bytes each" FORCE)
add_executable(picoAlpha src/board.cpp ${PICOALPHA_NET_SOURCE} src/mcts.cpp src/pico_main.cpp)
# Generated at configure time: avoid quoted, space-containing -D flags on Windows.
configure_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/include/pico_build_info.hpp.in"
  "${CMAKE_CURRENT_BINARY_DIR}/generated/pico_build_info.hpp"
  @ONLY
)

# Master diagnostic gate: OFF even if older CMake caches retain group flags ON.
option(PA_TELEM_DETAIL "Enable optional diagnostics (MCTS, NN, MEM)" OFF)
foreach(flag PA_TELEM_MCTS PA_TELEM_MEM PA_TELEM_NN PA_NN_PROFILE PA_NN_WORKSPACE PA_NN_FC_FAST PA_NN_FC_INT8 PA_NN_DSP PA_NN_CACHE PA_MCTS_REUSE PA_DUAL_CORE_NN)
 if(flag MATCHES "^PA_TELEM_(MCTS|MEM|NN)$" AND NOT PA_TELEM_DETAIL)
  target_compile_definitions(picoAlpha PRIVATE ${flag}=0)
 elseif(${flag})
  target_compile_definitions(picoAlpha PRIVATE ${flag}=1)
 else()
  target_compile_definitions(picoAlpha PRIVATE ${flag}=0)
 endif()
 message(STATUS "picoAlpha ${flag}: ${${flag}}")
endforeach()
# PA_NN_OPT_END
if(PA_SEGMENTED_TREE)
 target_compile_definitions(picoAlpha PRIVATE PA_SEGMENTED_TREE=1 PA_BUILD_SEGMENTED_CONFIRMED=1)
else()
 target_compile_definitions(picoAlpha PRIVATE PA_SEGMENTED_TREE=0 PA_BUILD_ALLOW_LEGACY=1)
endif()
target_compile_definitions(picoAlpha PRIVATE PA_STATIC_TREE_NODES=${PA_STATIC_TREE_NODES})
target_include_directories(picoAlpha PRIVATE include config "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(picoAlpha PRIVATE pico_stdlib pico_cyw43_arch_lwip_poll pico_multicore)
target_compile_definitions(picoAlpha PRIVATE PICO_CYW43_ARCH_POLL=1)
target_compile_options(picoAlpha PRIVATE -O3 -ffunction-sections -fdata-sections)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/config.h")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/include/config.h" _mdns REGEX "^[ \t]*inline[ \t]+constexpr[ \t]+bool[ \t]+MDNS_ENABLED[ \t]*=")
if(_mdns MATCHES "MDNS_ENABLED[ \t]*=[ \t]*true")
 target_link_libraries(picoAlpha PRIVATE pico_lwip_mdns)
 target_compile_definitions(picoAlpha PRIVATE PICOSARGON_MDNS_COMPILED=1)
else()
 target_compile_definitions(picoAlpha PRIVATE PICOSARGON_MDNS_COMPILED=0)
endif()
pico_enable_stdio_usb(picoAlpha 1)
pico_enable_stdio_uart(picoAlpha 0)
pico_add_extra_outputs(picoAlpha)
