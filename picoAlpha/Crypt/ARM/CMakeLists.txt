cmake_minimum_required(VERSION 3.16)
include(pico_sdk_import.cmake)
if(NOT DEFINED PICO_BOARD)
 set(PICO_BOARD pico2_w CACHE STRING "Target board" FORCE)
endif()
if(NOT PICO_BOARD STREQUAL "pico2_w" AND NOT PICO_BOARD STREQUAL "pico_w")
 message(FATAL_ERROR "Supported boards: pico2_w (default) or pico_w (experimental)")
endif()
project(picoAlpha_firmware VERSION 0.2.75 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PICO_CXX_ENABLE_EXCEPTIONS 1)
pico_sdk_init()
if(PICO_BOARD STREQUAL "pico_w")
 message(WARNING "Pico W experimental: verificar margen de SRAM, heap y MCTS antes de uso prolongado.")
 add_compile_definitions(PICOALPHA_TARGET_PICO_W=1)
endif()
# v0.2.73: picoGyal 1x8 is the default even when an existing CMake cache
# still contains PA_FORCE_PICOGYAL=OFF and PICOALPHA_NET=2x16 from v0.2.72.
# Opt in to the original TinyGyal 2x16 with -DPA_FORCE_TINYGYAL=ON.
set(PICOALPHA_NET "1x8" CACHE STRING "Network selected at compile time: 1x8 or 2x16")
set_property(CACHE PICOALPHA_NET PROPERTY STRINGS 1x8 2x16)
option(PA_FORCE_PICOGYAL "Legacy option: select picoGyal 1x8" ON)
option(PA_FORCE_TINYGYAL "Explicitly select original TinyGyal 2x16" OFF)
if(PA_FORCE_TINYGYAL)
 set(PICOALPHA_NET 2x16 CACHE STRING "Network" FORCE)
else()
 set(PICOALPHA_NET 1x8 CACHE STRING "Network" FORCE)
endif()
option(EPE "Enable pre-emptive MCTS during opponent thinking" ON)
option(PA_EPE_CLASSIC_GUIDE "Lightweight Sargon Classic root guidance during EPE only" ON)
set(PA_EPE_GUIDE_TOPK 6 CACHE STRING "Classic-guided root branches (1..32)")
if(PA_EPE_GUIDE_TOPK LESS 1 OR PA_EPE_GUIDE_TOPK GREATER 32)
 message(FATAL_ERROR "PA_EPE_GUIDE_TOPK must be 1..32")
endif()
set(PA_EPE_MAX_MS 30000 CACHE STRING "Maximum speculative compute duration per opponent turn")
set(PA_EPE_SIMULATIONS 400 CACHE STRING "Maximum speculative MCTS simulations")

if(PICOALPHA_NET STREQUAL "1x8")
  set(PICOALPHA_NET_SOURCE src/net_1x8.cpp)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/picoGyal.hpp" OR
     NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/picoGyal_fc_int8.hpp")
    message(FATAL_ERROR "Missing 1x8 INT8 coefficients: run tools/picoDistiller/picoExport.py first")
  endif()
elseif(PICOALPHA_NET STREQUAL "2x16")
  set(PICOALPHA_NET_SOURCE src/net.cpp)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/generated_tinygyal.hpp")
    message(FATAL_ERROR "Missing original TinyGyal 2x16 coefficients")
  endif()
else()
  message(FATAL_ERROR "Invalid PICOALPHA_NET: ${PICOALPHA_NET}")
endif()
message(STATUS "picoAlpha network: ${PICOALPHA_NET}")
# PA_SEGMENTED_TREE_BEGIN — v0.2.20: fail-closed, segmented by default.
# NEW rollback option deliberately avoids any stale PA_SEGMENTED_TREE=OFF cache.
option(PA_LEGACY_MCTS_TREE "Explicit rollback to monolithic tree (unsafe for long matches)" OFF)
if(PA_LEGACY_MCTS_TREE)
 set(PA_SEGMENTED_TREE OFF CACHE BOOL "Derived, do not edit" FORCE)
else()
 set(PA_SEGMENTED_TREE ON CACHE BOOL "Derived, do not edit" FORCE)
endif()
message(STATUS "picoAlpha v${PROJECT_VERSION} MCTS_STORAGE=${PA_SEGMENTED_TREE} (PA_LEGACY_MCTS_TREE=${PA_LEGACY_MCTS_TREE})")
# PA_SEGMENTED_TREE_END
# PA_NN_OPT_BEGIN: switchable network paths, default conservative FP32.
option(PA_TELEM_MCTS "MCTS guards and integrity telemetry" ON)
option(PA_TELEM_MEM "Detailed fixed arena and growth telemetry" ON)
option(PA_TELEM_NN "Neural compute telemetry" ON)
option(PA_NN_PROFILE "Profile TinyGyal layers" ON)
option(PA_NN_WORKSPACE "Reusable TinyGyal buffers" ON)
option(PA_NN_FC_FAST "4-lane FP32 fully-connected kernel" ON)
option(PA_NN_FC_INT8 "Quantized INT8 FC1" ON)
set(PA_NN_FC_INT8 ON CACHE BOOL "FC1 INT8 default" FORCE)
option(PA_NN_DSP "Use Cortex-M DSP dot products when supported" ON)
option(PA_NN_CACHE "8-slot history-exact neural inference cache" ON)
option(PA_MCTS_REUSE "Reuse validated MCTS descendants across moves" ON)
if(EPE AND NOT PA_MCTS_REUSE)
 message(FATAL_ERROR "EPE requires PA_MCTS_REUSE=ON")
endif()
# v0.2.66: explicit dual-core default, independent of v0.2.63 cached rollback.
set(PA_CORE1_STACK_BYTES 16384 CACHE STRING "Dedicated core 1 stack bytes")
option(PA_FORCE_SINGLE_CORE "Emergency fallback: core 0 only" OFF)
if(PA_FORCE_SINGLE_CORE)
  set(PA_DUAL_CORE_NN OFF CACHE BOOL "RP2350 dual-core inference" FORCE)
else()
  set(PA_DUAL_CORE_NN ON CACHE BOOL "RP2350 dual-core inference" FORCE)
endif()
set(PA_RECOVERY_SINGLE_CORE ${PA_FORCE_SINGLE_CORE} CACHE BOOL "Deprecated recovery alias" FORCE)
option(PA_FC1_PREPACK "Four-row tile packed FC1 kernel" ON)
option(PA_FC1_DSP_PACKED "Use packed operand extraction for SMLAD" ON)
option(PA_FC1_FLASH_PREFETCH "Prefetch subsequent FC1 weight tiles in XIP Flash" ON)
set(PA_FC1_CORE0_GROUPS 17 CACHE STRING "FC1 4-output groups assigned to core 0 (1..31)")
if(PA_FC1_CORE0_GROUPS LESS 1 OR PA_FC1_CORE0_GROUPS GREATER 31)
 message(FATAL_ERROR "PA_FC1_CORE0_GROUPS must be 1..31")
endif()
option(PA_BINARY_FIRST_CONV "Direct 112x8x8 sparse/binary convolution" ON)
option(PA_INT8_ACTIVATIONS "Static-scale INT8 residual convolution and SE path" ON)
option(PA_CALIBRATION_TELEMETRY "Intensive activation histograms (not for timing benchmarks)" OFF)
option(PA_FC1_PREPACK "Four-row tile packed FC1 kernel" ON)
set(PA_STATIC_TREE_NODES 7500 CACHE STRING "MCTS static SRAM nodes, 20 bytes each; reserve RAM for dual-core stack" FORCE)
add_executable(picoAlpha src/board.cpp ${PICOALPHA_NET_SOURCE} src/mcts.cpp src/pico_main.cpp)
target_compile_definitions(picoAlpha PRIVATE PA_COMPILED_IS_1X8=$<STREQUAL:${PICOALPHA_NET},1x8> EPE=$<BOOL:${EPE}> PA_EPE_MAX_MS=${PA_EPE_MAX_MS} PA_EPE_SIMULATIONS=${PA_EPE_SIMULATIONS} PA_EPE_CLASSIC_GUIDE=$<BOOL:${PA_EPE_CLASSIC_GUIDE}> PA_EPE_GUIDE_TOPK=${PA_EPE_GUIDE_TOPK})
target_compile_definitions(picoAlpha PRIVATE PA_FC1_CORE0_GROUPS=${PA_FC1_CORE0_GROUPS} PA_FC1_DSP_PACKED=$<BOOL:${PA_FC1_DSP_PACKED}> PA_FC1_FLASH_PREFETCH=$<BOOL:${PA_FC1_FLASH_PREFETCH}>)
# Generated at configure time: avoid quoted, space-containing -D flags on Windows.
configure_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/include/pico_build_info.hpp.in"
  "${CMAKE_CURRENT_BINARY_DIR}/generated/pico_build_info.hpp"
  @ONLY
)

# Master diagnostic gate: OFF even if older CMake caches retain group flags ON.
option(PA_TELEM_DETAIL "Enable optional diagnostics (MCTS, NN, MEM)" ON)
option(PA_DIAGNOSTIC_RUN "Instrument each search for bottleneck analysis regardless of cached telemetry options" ON)
option(PA_NN_TRACE "Verbose per-stage serial trace (emergency only)" OFF)
option(PA_CALIBRATION_RUN "Enable expensive activation histograms" OFF)
set(PA_CALIBRATION_TELEMETRY ${PA_CALIBRATION_RUN} CACHE BOOL "Activation histograms" FORCE)
target_compile_definitions(picoAlpha PRIVATE PA_NN_TRACE=$<BOOL:${PA_NN_TRACE}>)
foreach(flag PA_TELEM_MCTS PA_TELEM_MEM PA_TELEM_NN PA_NN_WORKSPACE PA_NN_FC_FAST PA_NN_FC_INT8 PA_NN_DSP PA_NN_CACHE PA_MCTS_REUSE PA_DUAL_CORE_NN PA_BINARY_FIRST_CONV PA_INT8_ACTIVATIONS PA_FC1_PREPACK PA_CALIBRATION_TELEMETRY)
 if(PA_DIAGNOSTIC_RUN AND flag MATCHES "^PA_TELEM_(MCTS|MEM|NN)$")
  target_compile_definitions(picoAlpha PRIVATE ${flag}=1)
 elseif(flag MATCHES "^PA_TELEM_(MCTS|MEM|NN)$" AND NOT PA_TELEM_DETAIL)
  target_compile_definitions(picoAlpha PRIVATE ${flag}=0)
 elseif(${flag})
  target_compile_definitions(picoAlpha PRIVATE ${flag}=1)
 else()
  target_compile_definitions(picoAlpha PRIVATE ${flag}=0)
 endif()
 message(STATUS "picoAlpha ${flag}: ${${flag}} (diagnostic=${PA_DIAGNOSTIC_RUN})")
endforeach()
if(PA_DIAGNOSTIC_RUN)
 target_compile_definitions(picoAlpha PRIVATE PA_DIAGNOSTIC_RUN=1 PA_NN_PROFILE=1)
else()
 target_compile_definitions(picoAlpha PRIVATE PA_DIAGNOSTIC_RUN=0 PA_NN_PROFILE=$<BOOL:${PA_NN_PROFILE}>)
endif()
message(STATUS "picoAlpha RECOVERY_SINGLE_CORE=${PA_RECOVERY_SINGLE_CORE} DUAL_CORE_NN=${PA_DUAL_CORE_NN}")
# PA_NN_OPT_END
if(PA_SEGMENTED_TREE)
 target_compile_definitions(picoAlpha PRIVATE PA_SEGMENTED_TREE=1 PA_BUILD_SEGMENTED_CONFIRMED=1)
else()
 target_compile_definitions(picoAlpha PRIVATE PA_SEGMENTED_TREE=0 PA_BUILD_ALLOW_LEGACY=1)
endif()
target_compile_definitions(picoAlpha PRIVATE PA_STATIC_TREE_NODES=${PA_STATIC_TREE_NODES} PA_CORE1_STACK_BYTES=${PA_CORE1_STACK_BYTES})
target_include_directories(picoAlpha PRIVATE include config "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(picoAlpha PRIVATE pico_stdlib pico_cyw43_arch_lwip_poll pico_multicore)
target_compile_definitions(picoAlpha PRIVATE PICO_CYW43_ARCH_POLL=1)
target_compile_options(picoAlpha PRIVATE -O3 -ffunction-sections -fdata-sections)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/config.h")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/include/config.h" _mdns REGEX "^[ \t]*inline[ \t]+constexpr[ \t]+bool[ \t]+MDNS_ENABLED[ \t]*=")
if(_mdns MATCHES "MDNS_ENABLED[ \t]*=[ \t]*true")
 target_link_libraries(picoAlpha PRIVATE pico_lwip_mdns)
 target_compile_definitions(picoAlpha PRIVATE PICOSARGON_MDNS_COMPILED=1)
else()
 target_compile_definitions(picoAlpha PRIVATE PICOSARGON_MDNS_COMPILED=0)
endif()
pico_enable_stdio_usb(picoAlpha 1)
pico_enable_stdio_uart(picoAlpha 0)
pico_add_extra_outputs(picoAlpha)
