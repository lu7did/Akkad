cmake_minimum_required(VERSION 3.16)
project(picoAlpha_cpp VERSION 0.1.42 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(picoAlpha src/main.cpp src/board.cpp src/nnue.cpp src/mcts.cpp)
target_include_directories(picoAlpha PRIVATE src)
target_link_libraries(picoAlpha PRIVATE Threads::Threads)
if(MSVC)
 target_compile_options(picoAlpha PRIVATE /W4 /EHsc /utf-8 /O2)
else()
 target_compile_options(picoAlpha PRIVATE -Wall -Wextra -Wpedantic -O3)
endif()
set_target_properties(picoAlpha PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 string(TOUPPER ${CONFIG} UC)
 set_target_properties(picoAlpha PROPERTIES "RUNTIME_OUTPUT_DIRECTORY_${UC}" "${CMAKE_BINARY_DIR}")
endforeach()
