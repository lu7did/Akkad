cmake_minimum_required(VERSION 3.16)
project(picoSargon VERSION 0.2.20 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Este build corresponde a Debian/Linux")
endif()
option(STEROIDS "Enable aspiration windows, SEE and razoring" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SHARED_HOST "${CMAKE_CURRENT_SOURCE_DIR}/../../x86/picoSargon")
add_executable(picoSargon "${SHARED_HOST}/src/host_main.cpp" "${SHARED_HOST}/src/board.cpp" "${SHARED_HOST}/src/engine.cpp" "${SHARED_HOST}/src/grumpy.cpp")
target_include_directories(picoSargon PRIVATE "${SHARED_HOST}/include")
if(STEROIDS)
 target_compile_definitions(picoSargon PRIVATE PICOSARGON_STEROIDS=1)
endif()
target_compile_options(picoSargon PRIVATE -Wall -Wextra -Wno-psabi)
set_target_properties(picoSargon PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
message(STATUS "Linux architecture=${CMAKE_SYSTEM_PROCESSOR}; STEROIDS=${STEROIDS}")
