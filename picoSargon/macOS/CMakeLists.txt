cmake_minimum_required(VERSION 3.16)
project(picoSargon VERSION 0.2.45 LANGUAGES CXX)
if(NOT APPLE)
 message(FATAL_ERROR "Este build corresponde exclusivamente a macOS")
endif()
option(STEROIDS "Enable experimental search optimizations" ON)
option(MULTICORE "Enable parallel root search on host platforms" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SHARED_HOST "${CMAKE_CURRENT_SOURCE_DIR}/../x86")
add_executable(picoSargon
 "${SHARED_HOST}/src/host_main.cpp"
 "${SHARED_HOST}/src/board.cpp"
 "${SHARED_HOST}/src/engine.cpp"
 "${SHARED_HOST}/src/grumpy.cpp")
target_include_directories(picoSargon PRIVATE "${SHARED_HOST}/include")
target_compile_definitions(picoSargon PRIVATE PICOSARGON_VERSION="${PROJECT_VERSION}")
if(STEROIDS)
 target_compile_definitions(picoSargon PRIVATE PICOSARGON_STEROIDS=1)
endif()
target_compile_options(picoSargon PRIVATE -Wall -Wextra)
set_target_properties(picoSargon PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
message(STATUS "macOS architecture=${CMAKE_SYSTEM_PROCESSOR}; STEROIDS=${STEROIDS}")

if(MULTICORE)
 find_package(Threads REQUIRED)
 target_link_libraries(picoSargon PRIVATE Threads::Threads)
 target_compile_definitions(picoSargon PRIVATE PICOSARGON_MULTICORE=1)
endif()
message(STATUS "picoSargon MULTICORE=${MULTICORE}")
