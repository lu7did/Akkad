cmake_minimum_required(VERSION 3.16)
project(picoSargon_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(picoSargon_host ../src/host_main.cpp ../src/board.cpp ../src/engine.cpp ../src/grumpy.cpp)
target_include_directories(picoSargon_host PRIVATE ../include)
if(MSVC)
    target_compile_options(picoSargon_host PRIVATE /W3)
else()
    target_compile_options(picoSargon_host PRIVATE -Wall -Wextra -Wno-psabi)
endif()
