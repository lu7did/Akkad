cmake_minimum_required(VERSION 3.16)
project(picoSargon_host VERSION 0.2.50 LANGUAGES CXX)
option(STEROIDS "Enable search optimizations" ON)
option(MULTICORE "Enable parallel root search on host platforms" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(picoSargon_host ../src/host_main.cpp ../src/board.cpp ../src/engine.cpp ../src/grumpy.cpp)
target_include_directories(picoSargon_host PRIVATE ../include)
target_compile_definitions(picoSargon_host PRIVATE PICOSARGON_VERSION="${PROJECT_VERSION}")
if(STEROIDS)
 target_compile_definitions(picoSargon_host PRIVATE PICOSARGON_STEROIDS=1)
endif()
message(STATUS "picoSargon STEROIDS=${STEROIDS}")
if(MSVC)
    target_compile_options(picoSargon_host PRIVATE /W3)
else()
    target_compile_options(picoSargon_host PRIVATE -Wall -Wextra -Wno-psabi)
endif()

if(MULTICORE)
 find_package(Threads REQUIRED)
 target_link_libraries(picoSargon_host PRIVATE Threads::Threads)
 target_compile_definitions(picoSargon_host PRIVATE PICOSARGON_MULTICORE=1)
endif()
message(STATUS "picoSargon MULTICORE=${MULTICORE}")
