cmake_minimum_required(VERSION 3.16)
project(tinyAlpha LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ZLIB REQUIRED)
add_executable(tinyAlpha src/main.cpp src/net.cpp src/board.cpp)
target_include_directories(tinyAlpha PRIVATE include)
target_link_libraries(tinyAlpha PRIVATE ZLIB::ZLIB)
if(MSVC)
 target_compile_options(tinyAlpha PRIVATE /O2 /W3)
else()
 target_compile_options(tinyAlpha PRIVATE -O3 -Wall -Wextra)
endif()

# Two executables allow the existing UCI adapter to benchmark FP32 vs INT8
# without any changes to brains.py, factory or the Python player.
add_executable(tinyAlphaInt8 src/main.cpp src/net.cpp src/board.cpp)
target_include_directories(tinyAlphaInt8 PRIVATE include)
target_link_libraries(tinyAlphaInt8 PRIVATE ZLIB::ZLIB)
target_compile_definitions(tinyAlphaInt8 PRIVATE TINYALPHA_DEFAULT_INT8=1)
if(MSVC)
 target_compile_options(tinyAlphaInt8 PRIVATE /O2 /W3)
else()
 target_compile_options(tinyAlphaInt8 PRIVATE -O3 -Wall -Wextra)
endif()
