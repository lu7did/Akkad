cmake_minimum_required(VERSION 3.20)
project(picoKeras VERSION 0.2.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(X86_SRC "${CMAKE_CURRENT_LIST_DIR}/../x86/picoKeras")
if(NOT EXISTS "${X86_SRC}/src/main.cpp")
  message(FATAL_ERROR "Missing picoKeras x86 source: ${X86_SRC}")
endif()
set(TFLITE_ROOT "" CACHE PATH "TensorFlow source checkout")
if(NOT EXISTS "${TFLITE_ROOT}/tensorflow/lite/CMakeLists.txt")
  message(FATAL_ERROR "Specify -DTFLITE_ROOT=<tensorflow source checkout>")
endif()
set(TFLITE_ENABLE_XNNPACK OFF CACHE BOOL "Lower first build resource usage" FORCE)
set(TFLITE_ENABLE_GPU OFF CACHE BOOL "No GPU delegate on Raspberry Pi" FORCE)
set(TFLITE_ENABLE_RUY ON CACHE BOOL "CPU kernels")
set(TFLITE_ENABLE_INSTALL OFF CACHE BOOL "No install target needed" FORCE)
add_subdirectory("${TFLITE_ROOT}/tensorflow/lite" "${CMAKE_CURRENT_BINARY_DIR}/tflite_build" EXCLUDE_FROM_ALL)
add_executable(picoKeras "${X86_SRC}/src/main.cpp" "${X86_SRC}/src/board.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/model_tflite.cpp")
target_include_directories(picoKeras PRIVATE "${X86_SRC}/include")
target_link_libraries(picoKeras PRIVATE tensorflow-lite)
target_compile_options(picoKeras PRIVATE -O2 -Wall -Wextra -Wpedantic)
