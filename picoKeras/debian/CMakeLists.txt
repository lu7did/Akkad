cmake_minimum_required(VERSION 3.20)
project(picoKeras VERSION 0.2.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PK_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(PK_SRC "${PK_ROOT}/src")
set(TFLITE_ROOT "" CACHE PATH "TensorFlow source checkout")
if(NOT EXISTS "${TFLITE_ROOT}/tensorflow/lite/CMakeLists.txt")
 message(FATAL_ERROR "-DTFLITE_ROOT must point to a TensorFlow checkout with tensorflow/lite/CMakeLists.txt")
endif()
set(TFLITE_ENABLE_XNNPACK OFF CACHE BOOL "Lower first build resource usage")
set(TFLITE_ENABLE_GPU OFF CACHE BOOL "No GPU delegate")
set(TFLITE_ENABLE_RUY ON CACHE BOOL "CPU kernels")
set(TFLITE_ENABLE_INSTALL OFF CACHE BOOL "No install target")
add_subdirectory("${TFLITE_ROOT}/tensorflow/lite" "${CMAKE_CURRENT_BINARY_DIR}/tflite_build" EXCLUDE_FROM_ALL)
add_executable(picoKeras "${PK_SRC}/main.cpp" "${PK_SRC}/board.cpp" "${PK_SRC}/model_tflite.cpp")
target_include_directories(picoKeras PRIVATE "${PK_SRC}")
target_link_libraries(picoKeras PRIVATE tensorflow-lite)
target_compile_definitions(picoKeras PRIVATE PICOKERAS_PLATFORM_DEBIAN=1 PICOKERAS_MODEL_TFLITE=1)
target_compile_options(picoKeras PRIVATE -O2 -Wall -Wextra -Wpedantic)
